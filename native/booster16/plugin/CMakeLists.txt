cmake_minimum_required(VERSION 3.22)
project(BoosterK1Policy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT SIMROBOT_SOURCE)
  message(FATAL_ERROR "Set SIMROBOT_SOURCE to pinned patched upstream source")
endif()
add_library(BoosterK1Policy MODULE BoosterK1Policy.cpp)
target_include_directories(BoosterK1Policy PRIVATE "${SIMROBOT_SOURCE}/Src/SimRobot" "${SIMROBOT_SOURCE}/Src/SimRobotCore3")
target_link_libraries(BoosterK1Policy PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_options(BoosterK1Policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(booster_policy_process_test process_test.cpp)
target_link_libraries(booster_policy_process_test PRIVATE Qt6::Core)
target_compile_options(booster_policy_process_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(booster_policy_lifecycle_test lifecycle_test.cpp)
target_compile_options(booster_policy_lifecycle_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(booster_policy_schedule_test schedule_test.cpp)
target_compile_options(booster_policy_schedule_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
