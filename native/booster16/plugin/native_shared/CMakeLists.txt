cmake_minimum_required(VERSION 3.22)
project(BoosterK1Native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
find_package(Python3 3.10 REQUIRED COMPONENTS Development)
if(NOT SIMROBOT_SOURCE)
  message(FATAL_ERROR "Set SIMROBOT_SOURCE to pinned patched9deb6ae6 source")
endif()
add_library(BoosterK1Native MODULE ../BoosterK1Policy.cpp)
target_compile_definitions(BoosterK1Native PRIVATE BOOSTER_NATIVE_SHARED)
target_include_directories(BoosterK1Native PRIVATE "${SIMROBOT_SOURCE}/Src/SimRobot" "${SIMROBOT_SOURCE}/Src/SimRobotCore3")
target_link_libraries(BoosterK1Native PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Python3::Python)
target_compile_options(BoosterK1Native PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(native_core_test native_core_test.cpp)
target_link_libraries(native_core_test PRIVATE Qt6::Core Python3::Python)
target_compile_options(native_core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
