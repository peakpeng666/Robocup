cmake_minimum_required(VERSION 3.22)
project(SoccerVideoObserver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT SIMROBOT_SOURCE)
 message(FATAL_ERROR "Set SIMROBOT_SOURCE to pinned SimRobot source")
endif()
add_library(SoccerVideoObserver MODULE SoccerVideoObserver.cpp)
target_include_directories(SoccerVideoObserver PRIVATE "${SIMROBOT_SOURCE}/Src/SimRobot" "${SIMROBOT_SOURCE}/Src/SimRobotCore3")
target_link_libraries(SoccerVideoObserver PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_options(SoccerVideoObserver PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_library(OfficialWalkVideoObserver MODULE OfficialWalkVideoObserver.cpp)
target_include_directories(OfficialWalkVideoObserver PRIVATE "${SIMROBOT_SOURCE}/Src/SimRobot" "${SIMROBOT_SOURCE}/Src/SimRobotCore3")
target_link_libraries(OfficialWalkVideoObserver PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_options(OfficialWalkVideoObserver PRIVATE -Wall -Wextra -Wpedantic -Werror)
